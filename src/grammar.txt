<Program>      ::= <Statement>* EOF
<Statement>    ::= <Command> | <Loop> | <Assignment> | <IfStatement>

<IfStatement>  ::= "IF" <Expression> <ComparisonOp> <Expression> "[" <Statement>* "]"
<ComparisonOp> ::= "==" | "<" | ">"

<Assignment>   ::= "SET" <Identifier> <Expression>

<Command>      ::= "FORWARD" <Expression>
                 | "BACKWARD" <Expression>
                 | "LEFT" <Expression>
                 | "RIGHT" <Expression>
                 | "PENUP" 
                 | "PENDOWN"
                 | "COLOR" <Expression>
                 | "WIDTH" <Expression>
                 | "SPEED" <Expression>

<Loop>         ::= "REPEAT" <Expression> "[" <Statement>* "]"

<Expression>   ::= <Term> (('+' | '-') <Term>)*
<Term>         ::= <Factor> (('*' | '/') <Factor>)*
<Factor>       ::= <Number> | <Identifier> | "(" <Expression> ")"
